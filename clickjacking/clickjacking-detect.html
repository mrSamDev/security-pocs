Embed this page in itself (same-origin demo)

The inner copy will detect that it is framed and paint a warning overlay. Same-origin framing lets us additionally inspect the parent's CSS.

Use on a different site

This file is the demo. To protect any other site, drop the reusable clickjacking-guard.js from this folder into that site and point at it from the <head>: 

<script>
  window.ClickjackGuardConfig = {
    bustOut: true,          // try to break out of the frame
    showOverlay: true,      // paint a warning when framed
    overlayText: 'This page cannot be displayed inside another site.',
    checkOwnHeaders: true,  // report X-Frame-Options / CSP of THIS page
    onDetect: function (r) { console.warn('clickjacked', r); }
  };
</script>
<script src="/clickjacking-guard.js" defer></script>

Or configure inline via data attributes: <script src="/clickjacking-guard.js" data-bust-out="false"></script>. Programmatic access: ClickjackGuard.run() / ClickjackGuard.report().

Zero-dependency inline guard (paste into a page)

<script>
/* Anti-clickjacking guard — runs on page load */
(function () {
  var framed;
  try { framed = window.self !== window.top; }
  catch (e) { framed = true; }              // cross-origin parent => framed
  if (!framed) return;

  // 1) Legacy frame-buster (works only when the parent is same-origin
  //    OR when the frame is not sandboxed).
  try { if (window.top.location.href !== window.self.location.href) {
          window.top.location = window.self.location; } } catch (e) {}

  // 2) Visible warning that cannot be silently ignored.
  document.addEventListener('DOMContentLoaded', function () {
    var o = document.createElement('div');
    o.style.cssText = 'position:fixed;inset:0;z-index:2147483647;' +
      'background:#7f1d1d;color:#fff;display:flex;align-items:center;' +
      'justify-content:center;font:700 18px sans-serif;text-align:center;padding:2rem';
    o.textContent = 'This page refused to be displayed inside another site.';
    document.body.appendChild(o);
  });
})();
</script>
